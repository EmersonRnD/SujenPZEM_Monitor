document.getElementById('btnConnect').onclick = async () => { try { document.getElementById('debug').innerText = "Checking existing permissions..."; // 1. Check if the device was previously authorized in Chrome if (navigator.bluetooth.getDevices) { const pairedDevices = await navigator.bluetooth.getDevices(); if (pairedDevices.length > 0) { bleDevice = pairedDevices[0]; document.getElementById('debug').innerText = "Reconnecting to known device: " + (bleDevice.name || "BLE Adapter"); } } // 2. If not previously authorized, open the native scanner without restrictive filters if (!bleDevice) { document.getElementById('debug').innerText = "Scanning for all BLE devices..."; bleDevice = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: [BLE_SERVICE_UUID, 0xFFE0, 0xFFE1, "0000ffe0-0000-1000-8000-00805f9b34fb"] }); } document.getElementById('debug').innerText = "Connecting to GATT Server..."; // Timeout guard to stop infinite spinning const connectTimeout = setTimeout(() => { throw new Error("GATT Connection timed out. Ensure device is unpaired from Android system settings."); }, 10000); bleServer = await bleDevice.gatt.connect(); clearTimeout(connectTimeout); document.getElementById('debug').innerText = "Discovering services..."; const service = await bleServer.getPrimaryService(BLE_SERVICE_UUID); bleChar = await service.getCharacteristic(BLE_CHAR_UUID); // Start listening to incoming data packets await bleChar.startNotifications(); bleChar.addEventListener('characteristicvaluechanged', handleNotifications); document.getElementById('btnConnect').innerText = "Connected: " + (bleDevice.name || "SH-B20"); document.getElementById('btnConnect').style.background = "#03dac6"; // Assert 300A shunt register await bleChar.writeValueWithoutResponse(shuntFrame300A); // Start 1-second polling loop clearInterval(pollInterval); pollInterval = setInterval(async () => { if (bleChar && bleServer && bleServer.connected) { try { await bleChar.writeValueWithoutResponse(reqFrame); } catch (e) { console.warn("Poll write dropped:", e); } } }, 1000); initLoggingTimer(); } catch (err) { bleDevice = null; document.getElementById('debug').innerText = "BLE Error: " + err.message; console.error(err); } };
